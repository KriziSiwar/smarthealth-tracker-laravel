pipeline {
    agent any

    environment {
        # Configuration de la DB pour les tests Laravel
        DB_CONNECTION = 'mysql'
        DB_HOST = '127.0.0.1'
        DB_PORT = '3306'
        DB_DATABASE = 'smarthealth-tracker'
        DB_USERNAME = 'root'          // Remplace par ton utilisateur MySQL
        DB_PASSWORD = ''       // Remplace par ton mot de passe MySQL
        APP_ENV = 'testing'
    }

    stages {
        stage('Checkout') {
            steps {
                echo 'Récupération du code...'
                checkout scm
            }
        }

        stage('Install Composer') {
            steps {
                echo 'Installation des dépendances PHP...'
                sh 'composer install --no-interaction --prefer-dist'
            }
        }

        stage('Install Node') {
            steps {
                echo 'Installation des dépendances front-end...'
                sh 'npm install'
                sh 'npm run build'
            }
        }

        stage('Prepare Database') {
            steps {
                echo 'Création et migration de la base de tests...'
                sh 'mysql -u $DB_USERNAME -p$DB_PASSWORD -e "CREATE DATABASE IF NOT EXISTS $DB_DATABASE;"'
                sh 'php artisan migrate --env=testing --force'
            }
        }

        stage('Run Tests') {
            steps {
                echo 'Exécution des tests Laravel...'
                sh 'php artisan test --env=testing'
            }
        }
    }

    post {
        always {
            echo 'Nettoyage post-build...'
        }
        success {
            echo 'Build et tests réussis !'
        }
        failure {
            echo 'Build ou tests échoués.'
        }
    }
}
