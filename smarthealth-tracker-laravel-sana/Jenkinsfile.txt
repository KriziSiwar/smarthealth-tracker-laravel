pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        VENV_DIR = "venv"
        PATH = "/usr/bin:/usr/local/bin:$PATH"
    }

    stages {

        stage('Checkout') {
            steps {
                echo 'Cloning repository...'
                git branch: 'integrationnn',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Install PHP & MySQL') {
            steps {
                echo 'Installing PHP, Composer, and MySQL...'
                sh '''
                sudo apt-get update
                sudo apt-get install -y php php-cli php-mbstring php-xml php-bcmath php-curl php-mysql unzip curl mysql-server

                # Start MySQL
                sudo service mysql start
                mysql -e "CREATE DATABASE IF NOT EXISTS laravel_test;"
                mysql -e "CREATE USER IF NOT EXISTS 'laravel'@'localhost' IDENTIFIED BY 'secret';"
                mysql -e "GRANT ALL PRIVILEGES ON laravel_test.* TO 'laravel'@'localhost';"
                mysql -e "FLUSH PRIVILEGES;"

                # Install Composer
                EXPECTED_SIGNATURE=$(curl -s https://composer.github.io/installer.sig)
                php -r "copy('https://getcomposer.org/installer', 'composer-setup.php');"
                php -r "if (hash_file('sha384', 'composer-setup.php') === '$EXPECTED_SIGNATURE') { echo 'Installer verified'; } else { echo 'Installer corrupt'; unlink('composer-setup.php'); } echo PHP_EOL;"
                php composer-setup.php --install-dir=/usr/local/bin --filename=composer
                php -r "unlink('composer-setup.php');"
                '''
            }
        }

        stage('Setup Laravel') {
            steps {
                echo 'Installing Laravel dependencies via Composer...'
                dir('smarthealth-tracker-laravel-sana/smarthealth-tracker-laravel') {
                    sh 'composer install --no-interaction --prefer-dist'
                    # Copier .env.testing pour configurer la DB
                    sh 'cp .env.example .env'
                    sh '''
                    php artisan key:generate
                    sed -i "s/DB_DATABASE=.*/DB_DATABASE=laravel_test/" .env
                    sed -i "s/DB_USERNAME=.*/DB_USERNAME=laravel/" .env
                    sed -i "s/DB_PASSWORD=.*/DB_PASSWORD=secret/" .env
                    '''
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo 'Installing Node dependencies and building frontend...'
                dir('smarthealth-tracker-laravel-sana/smarthealth-tracker-laravel') {
                    sh 'npm install'
                    sh 'npm run build'
                }
            }
        }

        stage('Laravel Tests') {
            steps {
                echo 'Running Laravel tests...'
                dir('smarthealth-tracker-laravel-sana/smarthealth-tracker-laravel') {
                    sh 'php artisan test --env=testing'
                }
            }
        }

        stage('Setup Flask AI') {
            steps {
                echo 'Setting up Python virtual environment for Flask AI...'
                sh """
                   python3 -m venv ${VENV_DIR}
                   source ${VENV_DIR}/bin/activate
                   pip install --upgrade pip
                   pip install -r flask_app/requirements.txt
                """
            }
        }

        stage('Run Flask Tests') {
            steps {
                echo 'Running Flask AI tests...'
                sh """
                   source ${VENV_DIR}/bin/activate
                   pytest flask_app/tests
                """
            }
        }

        stage('Deploy') {
            steps {
                echo 'Deploying project...'
                sh 'echo "Deployment placeholder"'
            }
        }
    }

    post {
        success {
            echo 'Pipeline finished successfully!'
        }
        failure {
            echo 'Pipeline failed!'
        }
    }
}
