pipeline {
    agent any

    tools {
        nodejs 'Node20' // Nom de NodeJS configuré dans Jenkins
    }

    environment {
        VENV_DIR = "venv"
        PATH = "/usr/local/bin:$PATH" // Assure que composer, php, npm sont trouvés
        COMPOSER_CMD = "/usr/bin/composer" // chemin absolu de composer sur ton système
    }

    stages {

        stage('Checkout') {
            steps {
                echo 'Cloning repository...'
                git branch: 'integrationnn',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Setup Laravel') {
            steps {
                echo 'Installing PHP dependencies via Composer...'
                sh "${COMPOSER_CMD} install --no-interaction --prefer-dist"
            }
        }

        stage('Build Frontend') {
            steps {
                echo 'Installing Node dependencies and building frontend...'
                sh 'npm install'
                sh 'npm run build'
            }
        }

        stage('Laravel Tests') {
            steps {
                echo 'Running Laravel tests...'
                sh 'php artisan test --verbose'
            }
        }

        stage('Setup Flask AI') {
            steps {
                echo 'Setting up Python virtual environment for Flask AI...'
                sh "python3 -m venv ${VENV_DIR}"
                sh """
                   . ${VENV_DIR}/bin/activate
                   pip install --upgrade pip
                   pip install -r flask_app/requirements.txt
                """
            }
        }

        stage('Run Flask Tests') {
            steps {
                echo 'Running Flask AI tests...'
                sh """
                   . ${VENV_DIR}/bin/activate
                   pytest flask_app/tests
                """
            }
        }

        stage('Deploy') {
            steps {
                echo 'Deploying project...'
                // Ici tu peux mettre tes commandes de déploiement réelles
                sh 'echo "Deployment placeholder"'
            }
        }
    }

    post {
        success {
            echo 'Pipeline finished successfully!'
        }
        failure {
            echo 'Pipeline failed!'
        }
    }
}
