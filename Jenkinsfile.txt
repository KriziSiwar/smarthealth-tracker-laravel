pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        VENV_DIR = "venv"
        LARAVEL_DIR = "smarthealth-tracker-laravel"
        FLASK_DIR = "appflask"
        DB_PATH = "database/database.sqlite"
    }

    stages {
        stage('Checkout') {
            steps {
                echo '🚀 Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Setup Laravel') {
            steps {
                echo '🔧 Setting up Laravel...'
                dir(env.LARAVEL_DIR) {
                    // Copy .env file
                    sh 'cp .env.example .env'
                    
                    // Install dependencies
                    sh 'composer install --no-interaction --prefer-dist --optimize-autoloader'
                    
                    // Generate app key
                    sh 'php artisan key:generate'
                    
                    // Update .env for testing
                    sh """
                        echo -e \"\\n# Testing Environment\" >> .env
                        echo \"APP_ENV=testing\" >> .env
                        echo \"DB_CONNECTION=sqlite\" >> .env
                        echo \"DB_DATABASE=$(pwd)/${env.DB_PATH}\" >> .env
                        echo \"CACHE_DRIVER=array\" >> .env
                        echo \"SESSION_DRIVER=array\" >> .env
                        echo \"QUEUE_DRIVER=sync\" >> .env
                    """
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo '🛠️ Building frontend assets...'
                dir(env.LARAVEL_DIR) {
                    sh 'npm install --no-audit --no-optional'
                    sh 'npm run build'
                }
            }
        }

        stage('Run Tests') {
            steps {
                echo '🧪 Running tests...'
                dir(env.LARAVEL_DIR) {
                    // Create database file
                    sh "touch ${env.DB_PATH}"
                    sh "chmod 666 ${env.DB_PATH}"
                    
                    // Run migrations and tests
                    sh 'php artisan migrate:fresh --force'
                    sh 'php artisan test'
                }
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up...'
            dir(env.LARAVEL_DIR) {
                sh "rm -f ${env.DB_PATH} || true"
            }
        }
        success {
            echo '✅ Pipeline completed successfully!'
        }
        failure {
            echo '❌ Pipeline failed!'
        }
    }
}
