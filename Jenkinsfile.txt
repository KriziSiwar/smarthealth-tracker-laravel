pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        VENV_DIR = 'venv'
        LARAVEL_DIR = 'smarthealth-tracker-laravel'
        FLASK_DIR = 'appflask'
        DB_PATH = 'database/database.sqlite'
    }

    stages {
        stage('Checkout') {
            steps {
                echo '🚀 Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Setup Laravel') {
                steps {
                    echo '🔧 Setting up Laravel...'
                    dir(env.LARAVEL_DIR) {
                        // Copy .env and install dependencies
                        sh 'cp .env.example .env'
                        sh 'composer install --no-interaction --prefer-dist --optimize-autoloader'
                        sh 'php artisan key:generate'
                        
                        // Install Breeze non-interactively
                        sh 'php artisan breeze:install blade --pest --ssr --dark'
                        
                        // Update .env
                        sh '''
                            echo "" >> .env
                            echo "# Testing Environment" >> .env
                            echo "APP_ENV=testing" >> .env
                            echo "DB_CONNECTION=sqlite" >> .env
                            echo "DB_DATABASE=$(pwd)/database/database.sqlite" >> .env
                            echo "CACHE_DRIVER=array" >> .env
                            echo "SESSION_DRIVER=array" >> .env
                            echo "QUEUE_DRIVER=sync" >> .env
                        '''
                    }
                }
            }
        stage('Setup Frontend') {
            steps {
                echo '🔄 Setting up frontend...'
                dir(env.LARAVEL_DIR) {
                    // Installer Node.js et npm
                    sh 'node -v'
                    sh 'npm -v'
                    
                    // Nettoyer et installer les dépendances
                    sh 'rm -rf node_modules/'
                    sh 'rm -f package-lock.json'
                    sh 'npm cache clean --force'
                    sh 'npm install --no-audit --prefer-offline'
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo '🛠️ Building frontend assets...'
                dir(env.LARAVEL_DIR) {
                    // Désactiver les mises à jour automatiques
                    sh 'export VITE_DISABLE_OPEN_BROWSER=true'
                    sh 'npm run build'
                }
            }
        }

        stage('Run Tests') {
            steps {
                echo '🧪 Running tests...'
                dir(env.LARAVEL_DIR) {
                    sh 'mkdir -p database'
                    sh 'touch database/database.sqlite'
                    sh 'chmod 666 database/database.sqlite'
                    sh 'php artisan migrate:fresh --force'
                    sh 'php artisan test'
                }
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up...'
            dir(env.LARAVEL_DIR) {
                sh 'rm -f database/database.sqlite || true'
            }
        }
        success {
            echo '✅ Pipeline completed successfully!'
        }
        failure {
            echo '❌ Pipeline failed!'
        }
    }
}


stage('Setup Python Environment') {
    steps {
        echo '🐍 Setting up Python environment...'
        dir(env.FLASK_DIR) {
            // Création d'un environnement virtuel
            sh 'python -m venv venv'
            sh 'source venv/bin/activate && pip install --upgrade pip'
            
            // Installation des dépendances
            sh 'source venv/bin/activate && pip install -r requirements.txt'
            
            // Installation des dépendances de développement
            sh 'source venv/bin/activate && pip install pytest pytest-cov'
        }
    }
}

stage('Run Python Tests') {
    steps {
        echo '🧪 Running Python tests...'
        dir(env.FLASK_DIR) {
            // Exécution des tests avec couverture
            sh 'source venv/bin/activate && python -m pytest --cov=app --cov-report=xml:coverage.xml'
            
            // Stocker le rapport de couverture
            junit '**/test-reports/*.xml'
            cobertura '**/coverage.xml'
        }
    }
}
