pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        VENV_DIR = "venv"
        PATH = "/usr/bin:/usr/local/bin:$PATH"
    }

    stages {

        stage('Checkout') {
            steps {
                echo 'Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Setup Laravel') {
            steps {
                echo 'Installing PHP dependencies via Composer...'
                dir('smarthealth-tracker-laravel') {
                    sh 'composer install --no-interaction --prefer-dist'
                }
            }
        }

        stage('Laravel Migrations') {
            steps {
                echo 'Preparing SQLite database and running migrations...'
                dir('smarthealth-tracker-laravel') {
                    sh 'mkdir -p database'
                    sh 'touch database/database.sqlite'
                    sh 'php artisan migrate:status'
                    sh 'php artisan migrate --force'
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo 'Installing Node dependencies and building frontend...'
                dir('smarthealth-tracker-laravel') {
                    sh 'npm install'
                    sh 'npm run build'
                }
            }
        }

        stage('Laravel Tests') {
            steps {
                echo 'Running Laravel tests...'
                dir('smarthealth-tracker-laravel') {
                    sh 'php artisan test'
                }
            }
        }

        stage('Setup Flask AI') {
            steps {
                echo 'Setting up Python virtual environment for Flask AI...'
                sh """
                   python3 -m venv ${VENV_DIR}
                   source ${VENV_DIR}/bin/activate
                   pip install --upgrade pip
                   pip install -r flask_app/requirements.txt
                """
            }
        }
