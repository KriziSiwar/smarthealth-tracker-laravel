pipeline {
    agent any

    environment {
        // Utilisez votre nom d'utilisateur Docker Hub
        DOCKER_IMAGE = "siwarkrizii/smarthealth-tracker:${BUILD_NUMBER}"
    }

    stages {
        stage('Checkout') {
            steps {
                echo '🚀 Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Build Docker Image') {
            steps {
                script {
                    echo '🐳 Building Docker image...'
                    sh "docker build -t ${DOCKER_IMAGE} ."
                }
            }
        }

        stage('Run Tests in Docker') {
            steps {
                script {
                    echo '🧪 Running tests in Docker container...'
                    sh "docker run --rm ${DOCKER_IMAGE} php artisan test"
                }
            }
        }

        stage('Login to Docker Hub') {
            when {
                branch 'main'
            }
            steps {
                withCredentials([usernamePassword(
                    credentialsId: 'docker-hub-credentials',
                    usernameVariable: 'DOCKER_USERNAME',
                    passwordVariable: 'DOCKER_PASSWORD'
                )]) {
                    sh '''
                        echo "$DOCKER_PASSWORD" | docker login -u "$DOCKER_USERNAME" --password-stdin
                    '''
                }
            }
        }

        stage('Push to Docker Hub') {
            when {
                branch 'main'
            }
            steps {
                script {
                    withCredentials([usernamePassword(
                        credentialsId: 'docker-hub-credentials',
                        usernameVariable: 'DOCKER_USERNAME',
                        passwordVariable: 'DOCKER_PASSWORD'
                    )]) {
                        // Tagger l'image
                        sh "docker tag ${DOCKER_IMAGE} ${DOCKER_IMAGE}"
                        sh "docker tag ${DOCKER_IMAGE} siwarkrizii/smarthealth-tracker:latest"
                        
                        // Pousser l'image
                        sh "docker push ${DOCKER_IMAGE}"
                        sh "docker push siwarkrizii/smarthealth-tracker:latest"
                    }
                }
            }
        }

        stage('Deploy with Docker Compose') {
            when {
                branch 'main'
            }
            steps {
                script {
                    // Arrêter et supprimer les conteneurs existants
                    sh 'docker-compose down || true'
                    
                    // Mettre à jour les images
                    sh 'docker-compose pull'
                    
                    // Démarrer les conteneurs
                    sh 'docker-compose up -d'
                    
                    // Attendre que les services soient prêts
                    sh 'sleep 30'
                    
                    // Exécuter les migrations
                    sh 'docker-compose exec -T app php artisan migrate --force'
                }
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up...'
            // Supprimer les conteneurs arrêtés
            sh 'docker container prune -f'
            // Supprimer les images non utilisées
            sh 'docker image prune -f'
        }
        success {
            echo '✅ Pipeline completed successfully!'
        }
        failure {
            echo '❌ Pipeline failed!'
        }
    }
}
