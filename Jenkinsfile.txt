pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        VENV_DIR = 'venv'
        LARAVEL_DIR = 'smarthealth-tracker-laravel'
        FLASK_DIR = 'appflask'
        DB_PATH = 'database/database.sqlite'
    }

    stages {
        stage('Checkout') {
            steps {
                echo '🚀 Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Setup Laravel') {
            steps {
                echo '🔧 Setting up Laravel...'
                dir(env.LARAVEL_DIR) {
                    // Copier .env et installer les dépendances
                    sh 'cp .env.example .env'
                    sh 'composer install --no-interaction --prefer-dist --optimize-autoloader'
                    sh 'php artisan key:generate'
                    
                    // Mise à jour du .env
                    sh '''
                        echo "" >> .env
                        echo "# Testing Environment" >> .env
                        echo "APP_ENV=testing" >> .env
                        echo "DB_CONNECTION=sqlite" >> .env
                        echo "DB_DATABASE=$(pwd)/database/database.sqlite" >> .env
                        echo "CACHE_DRIVER=array" >> .env
                        echo "SESSION_DRIVER=array" >> .env
                        echo "QUEUE_DRIVER=sync" >> .env
                    '''
                }
            }
        }

        stage('Install Frontend Dependencies') {
            steps {
                echo '📦 Installing frontend dependencies...'
                dir(env.LARAVEL_DIR) {
                    // Supprimer node_modules et package-lock.json s'ils existent
                    sh 'rm -rf node_modules package-lock.json || true'
                    // Installer les dépendances
                    sh 'npm ci --no-audit --prefer-offline'
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo '🛠️ Building frontend assets...'
                dir(env.LARAVEL_DIR) {
                    // Désactiver les mises à jour automatiques
                    sh 'export VITE_DISABLE_OPEN_BROWSER=true'
                    sh 'npm run build -- --mode=production'
                }
            }
        }

        stage('Run Tests') {
            steps {
                echo '🧪 Running tests...'
                dir(env.LARAVEL_DIR) {
                    sh 'mkdir -p database'
                    sh 'touch database/database.sqlite'
                    sh 'chmod 666 database/database.sqlite'
                    sh 'php artisan migrate:fresh --force'
                    sh 'php artisan test'
                }
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up...'
            dir(env.LARAVEL_DIR) {
                sh 'rm -f database/database.sqlite || true'
            }
        }
        success {
            echo '✅ Pipeline completed successfully!'
        }
        failure {
            echo '❌ Pipeline failed!'
        }
    }
}
