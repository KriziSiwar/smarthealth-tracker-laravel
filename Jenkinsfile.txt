pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        VENV_DIR = "venv"
        LARAVEL_DIR = "smarthealth-tracker-laravel"
        FLASK_DIR = "appflask"
        PATH = "/usr/bin:/usr/local/bin:$PATH"
    }

    stages {
        stage('Checkout') {
            steps {
                echo '🚀 Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Setup Laravel') {
            steps {
                echo '🔧 Setting up Laravel...'
                dir(env.LARAVEL_DIR) {
                    // Install PHP dependencies
                    sh 'composer install --no-interaction --prefer-dist --optimize-autoloader'
                    
                    // Copy environment file if not exists
                    sh 'test -f .env || cp .env.example .env'
                    
                    // Generate application key if not set
                    sh 'grep -q "^APP_KEY=" .env || php artisan key:generate'
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo '🛠️ Building frontend assets...'
                dir(env.LARAVEL_DIR) {
                    sh 'npm install'
                    sh 'npm run build'  // Changed from 'npx vite build' to use the proper npm script
                }
            }
        }

        stage('Run Laravel Tests') {
            environment {
                DB_CONNECTION = 'sqlite'
                DB_DATABASE = '${WORKSPACE}/${LARAVEL_DIR}/database/database.sqlite'
            }
            steps {
                echo '🧪 Running Laravel tests...'
                dir(env.LARAVEL_DIR) {
                    // Create SQLite database for testing
                    sh 'touch database/database.sqlite'
                    sh 'php artisan migrate:fresh --env=testing --force'
                    sh 'php artisan test --env=testing'
                }
            }
        }

        stage('Setup Flask AI') {
            steps {
                echo '🐍 Setting up Python virtual environment for Flask AI...'
                dir(env.FLASK_DIR) {
                    sh """
                        python3 -m venv ${env.VENV_DIR}
                        . ${env.VENV_DIR}/bin/activate
                        pip install --upgrade pip
                        pip install -r requirements.txt
                    """
                }
            }
        }

        stage('Run Flask Tests') {
            steps {
                echo '🧪 Running Flask AI tests...'
                dir(env.FLASK_DIR) {
                    sh """
                        . ${env.VENV_DIR}/bin/activate
                        python -m pytest tests -v
                    """
                }
            }
        }

        stage('Deploy') {
            when {
                branch 'main'  // Only deploy from main branch
            }
            steps {
                echo '🚀 Starting deployment...'
                // Add your deployment steps here
                // Example: sh 'rsync -avz . user@server:/path/to/deploy'
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up workspace...'
            cleanWs()
        }
        success {
            echo '✅ Pipeline completed successfully!'
            // Add notification here (Slack, Email, etc.)
        }
        failure {
            echo '❌ Pipeline failed!'
            // Add failure notification here
        }
    }
}
