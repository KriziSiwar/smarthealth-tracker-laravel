pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        LARAVEL_DIR = 'smarthealth-tracker-laravel'
        DOCKER_IMAGE = "krizisiwar/smarthealth-tracker:${BUILD_NUMBER}"

        NEXUS_URL = 'localhost:8082'
        NEXUS_REPO = 'docker-hosted'
        NEXUS_IMAGE = "${NEXUS_URL}/${NEXUS_REPO}/smarthealth-tracker"
  
    }

    stages {
        stage('Checkout') {
            steps {
                echo '🚀 Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

      stage('Setup Laravel') {
            steps {
                echo '🔧 Setting up Laravel...'
                dir(env.LARAVEL_DIR) {
                    sh 'cp .env.example .env'
                    sh 'composer install --no-interaction --prefer-dist --optimize-autoloader'
                    sh 'php artisan key:generate'
                    sh 'php artisan breeze:install blade --pest --ssr --dark'
                }
            }
        }

        stage('Setup Frontend') {
            steps {
                echo '🔄 Setting up frontend...'
                dir(env.LARAVEL_DIR) {
                    sh 'npm install --no-audit --prefer-offline'
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo '🛠️ Building frontend assets...'
                dir(env.LARAVEL_DIR) {
                    sh 'export VITE_DISABLE_OPEN_BROWSER=true'
                    sh 'npm run build'
                }
            }
        }

        stage('Run Tests') {
            steps {
                echo '🧪 Running tests...'
                dir(env.LARAVEL_DIR) {
                    sh 'touch database/database.sqlite'
                    sh 'chmod 666 database/database.sqlite'
                    sh 'php artisan migrate:fresh --force'
                    sh 'php artisan test'
                }
            }
        }


stage('Debug File Structure') {
    steps {
        echo '🔍 Debugging file structure...'
        dir(env.LARAVEL_DIR) {
            sh '''
                echo "Current directory: $(pwd)"
                echo "Files in directory:"
                ls -la
                echo "Checking for Dockerfile:"
                if [ -f "Dockerfile" ]; then
                    echo "✅ Dockerfile exists"
                    cat Dockerfile
                else
                    echo "❌ Dockerfile not found in $(pwd)"
                    echo "Searching for Dockerfile in parent directories:"
                    find .. -name "Dockerfile" -type f
                fi
            '''
        }
    }
}


 stage('Build and Push to Nexus') {
    environment {
        NEXUS_CREDS = credentials('nexus-credentials')
    }
    steps {
        echo '🏗️ Building and pushing to Nexus...'
        script {
            // Change to the repository root directory
            dir('.') {
                // Build the Docker image from the root
                def imageTag = "localhost:8082/smarthealth-tracker:${env.BUILD_NUMBER}"
                sh "docker build -t ${imageTag} -f Dockerfile ."
                
                // Login and push
                withCredentials([usernamePassword(
                    credentialsId: 'nexus-credentials',
                    usernameVariable: 'NEXUS_USER',
                    passwordVariable: 'NEXUS_PASS'
                )]) {
                    sh '''
                        echo "Testing Nexus connection..."
                        if ! curl -s -I http://localhost:8082/service/rest/v1/status; then
                            echo "ERROR: Cannot connect to Nexus"
                            exit 1
                        fi
                        
                        echo "Logging into Nexus Docker registry..."
                        echo "$NEXUS_PASS" | docker login -u "$NEXUS_USER" --password-stdin localhost:8082
                        
                        echo "Pushing image..."
                        docker push "localhost:8082/smarthealth-tracker:${BUILD_NUMBER}"
                    '''
                }
            }
        }
    }
}

        stage('Deploy from Nexus') {
            when {
                anyOf {
                    branch 'siwar_new_integration'
                    expression { 
                        def currentBranch = env.BRANCH_NAME ?: sh(
                            script: 'git rev-parse --abbrev-ref HEAD', 
                            returnStdout: true
                        ).trim()
                        return currentBranch == 'siwar_new_integration' || 
                               currentBranch == 'origin/siwar_new_integration'
                    }
                }
            }
            environment {
                NEXUS_CREDS = credentials('nexus-token')
            }
            steps {
                script {
                    def imageToDeploy = "${NEXUS_IMAGE}:${env.BUILD_NUMBER}"
                    
                    // Arrêter et supprimer le conteneur existant
                    sh 'docker stop smarthealth-app || true'
                    sh 'docker rm smarthealth-app || true'
                    
                    // Pull et run la nouvelle image
                    sh """
                        docker pull ${imageToDeploy}
                        docker run -d \
                            --name smarthealth-app \
                            -p 80:80 \
                            --restart unless-stopped \
                            ${imageToDeploy}
                    """
                }
            }
        }


    stage('SonarQube Analysis') {
    environment {
        SONAR_AUTH_TOKEN = credentials('sonarToken')
    }
    steps {
        echo '🔍 Running SonarQube analysis...'
        dir(env.LARAVEL_DIR) {
            sh 'npm install -g sonar-scanner'
            sh '''
                # Create sonar-project.properties
                cat > sonar-project.properties << 'EOL'
                sonar.projectKey=smarthealth-tracker
                sonar.projectName=SmartHealth Tracker
                sonar.projectVersion=1.0
                sonar.sources=.
                sonar.sourceEncoding=UTF-8
                sonar.excludes=**/node_modules/**,**/*.spec.js
                sonar.php.coverage.reportPaths=coverage.xml
                EOL
                
                # Run sonar-scanner with token
                sonar-scanner -Dsonar.host.url=http://localhost:9000 -Dsonar.login=$SONAR_AUTH_TOKEN
            '''
        }
    }
}

stage('Test SonarQube Connection') {
    environment {
        SONAR_AUTH_TOKEN = credentials('sonarToken')
    }
    steps {
        sh '''
            echo "Testing SonarQube connection..."
            curl -s -o /dev/null -w "HTTP Status: %{http_code}\n" \
            -u $SONAR_AUTH_TOKEN: http://localhost:9000/api/system/health
        '''
    }
}







        stage('Build Docker Image') {
    steps {
        script {
            echo '🐳 Building Docker image...'
            sh """
                docker build -t ${DOCKER_IMAGE} .
                # Add additional tag for latest
                docker tag ${DOCKER_IMAGE} siwarkrizii/smarthealth-tracker:latest
            """
        }
    }
}

        stage('Debug Branch Info') {
            steps {
                script {
                    echo "Current branch: ${env.BRANCH_NAME}"
                    echo "Git branch: ${sh(script: 'git rev-parse --abbrev-ref HEAD', returnStdout: true).trim()}"
                }
            }
        }

       stage('Push to Docker Hub') {
    when {
        anyOf {
            branch 'siwar_new_integration'
            expression {
                def currentBranch = env.BRANCH_NAME ?: sh(script: 'git rev-parse --abbrev-ref HEAD', returnStdout: true).trim()
                return currentBranch == 'siwar_new_integration' || currentBranch == 'origin/siwar_new_integration'
            }
        }
    }
    steps {
        script {
            echo "🚢 Pushing Docker image to Docker Hub..."
            withCredentials([usernamePassword(
                credentialsId: 'docker-hub-credentials',
                usernameVariable: 'DOCKER_USERNAME',
                passwordVariable: 'DOCKER_PASSWORD'
            )]) {
                sh '''
                    echo "Logging into Docker Hub..."
                    echo "$DOCKER_PASSWORD" | docker login -u "$DOCKER_USERNAME" --password-stdin
                    echo "Pushing image $DOCKER_IMAGE..."
                    docker push "$DOCKER_IMAGE"
                    docker logout
                '''
            }
        }
    }
}

        stage('Deploy') {
            when {
                anyOf {
                    branch 'siwar_new_integration'
                    expression { 
                        def currentBranch = env.BRANCH_NAME ?: sh(script: 'git rev-parse --abbrev-ref HEAD', returnStdout: true).trim()
                        return currentBranch == 'siwar_new_integration' || currentBranch == 'origin/siwar_new_integration'
                    }
                }
            }
            steps {
                script {
                    echo "🚀 Deploying application..."
                    // Your deployment commands here
                    sh 'echo "Deployment would happen here"'
                }
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up...'
            sh 'docker container prune -f || true'
            sh 'docker image prune -f || true'
        }
    }
}
