pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        LARAVEL_DIR = 'smarthealth-tracker-laravel'
        DOCKER_IMAGE = "siwarkrizii/smarthealth-tracker:${BUILD_NUMBER}"
    }

    stages {
        // ... (keep all your existing stages until 'Build Docker Image')

        stage('Push to Docker Hub') {
            when {
                branch 'siwar_new_integration'
            }
            steps {
                script {
                    // First, verify Docker is installed and running
                    sh 'docker --version || echo "Docker is not installed"'
                    
                    // Login to Docker Hub with proper error handling
                    withCredentials([usernamePassword(
                        credentialsId: 'docker-hub-credentials',
                        usernameVariable: 'DOCKER_USERNAME',
                        passwordVariable: 'DOCKER_PASSWORD'
                    )]) {
                        // Use a different approach for Docker login
                        sh '''
                            echo "Logging into Docker Hub..."
                            echo $DOCKER_PASSWORD | docker login -u $DOCKER_USERNAME --password-stdin || {
                                echo "Docker login failed"
                                exit 1
                            }
                            
                            echo "Pushing image ${DOCKER_IMAGE} to Docker Hub..."
                            docker push ${DOCKER_IMAGE} || {
                                echo "Failed to push image"
                                docker logout
                                exit 1
                            }
                            
                            echo "Successfully pushed image to Docker Hub"
                            docker logout
                        '''
                    }
                }
            }
        }

        stage('Deploy') {
            when {
                branch 'siwar_new_integration'
            }
            steps {
                script {
                    // Install Docker Compose if not present
                    sh '''
                        # Check if docker-compose is installed
                        if ! command -v docker-compose &> /dev/null; then
                            echo "docker-compose not found, installing..."
                            sudo curl -L "https://github.com/docker/compose/releases/latest/download/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose
                            sudo chmod +x /usr/local/bin/docker-compose
                        fi
                        
                        echo "Stopping any running containers..."
                        docker-compose down --remove-orphans || true
                        
                        echo "Starting containers..."
                        docker-compose up -d --build
                        
                        echo "Checking container status..."
                        docker ps
                    '''
                }
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up...'
            sh 'docker container prune -f || true'
            sh 'docker image prune -f || true'
        }
    }
}
