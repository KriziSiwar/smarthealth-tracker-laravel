pipeline {
    agent any

    tools {
        nodejs 'Node20'
    }

    environment {
        VENV_DIR = 'venv'
        LARAVEL_DIR = 'smarthealth-tracker-laravel'
        FLASK_DIR = 'appflask'
        DB_PATH = 'database/database.sqlite'
        DOCKER_IMAGE = "smarthealth-tracker:${BUILD_NUMBER}"
        DOCKER_REGISTRY = "your-dockerhub-username"
    }

    stages {
        stage('Checkout') {
            steps {
                echo '🚀 Cloning repository...'
                git branch: 'siwar_new_integration',
                    url: 'https://github.com/KriziSiwar/smarthealth-tracker-laravel.git',
                    credentialsId: 'GithubToken-laravel'
            }
        }

        stage('Setup Laravel') {
            steps {
                echo '🔧 Setting up Laravel...'
                dir(env.LARAVEL_DIR) {
                    sh 'cp .env.example .env'
                    sh 'composer install --no-interaction --prefer-dist --optimize-autoloader'
                    sh 'php artisan key:generate'
                    sh 'php artisan breeze:install blade --pest --ssr --dark'
                    
                    sh '''
                        echo "" >> .env
                        echo "# Testing Environment" >> .env
                        echo "APP_ENV=testing" >> .env
                        echo "DB_CONNECTION=sqlite" >> .env
                        echo "DB_DATABASE=$(pwd)/database/database.sqlite" >> .env
                        echo "CACHE_DRIVER=array" >> .env
                        echo "SESSION_DRIVER=array" >> .env
                        echo "QUEUE_DRIVER=sync" >> .env
                    '''
                }
            }
        }

        stage('Setup Frontend') {
            steps {
                echo '🔄 Setting up frontend...'
                dir(env.LARAVEL_DIR) {
                    sh 'node -v'
                    sh 'npm -v'
                    sh 'rm -rf node_modules/'
                    sh 'rm -f package-lock.json'
                    sh 'npm cache clean --force'
                    sh 'npm install --no-audit --prefer-offline'
                }
            }
        }

        stage('Build Frontend') {
            steps {
                echo '🛠️ Building frontend assets...'
                dir(env.LARAVEL_DIR) {
                    sh 'export VITE_DISABLE_OPEN_BROWSER=true'
                    sh 'npm run build'
                }
            }
        }

        stage('Run Tests') {
            steps {
                echo '🧪 Running tests...'
                dir(env.LARAVEL_DIR) {
                    sh 'mkdir -p database'
                    sh 'touch database/database.sqlite'
                    sh 'chmod 666 database/database.sqlite'
                    sh 'php artisan migrate:fresh --force'
                    sh 'php artisan test'
                }
            }
        }

        stage('Build Docker Image') {
            steps {
                script {
                    echo '🐳 Building Docker image...'
                    sh "docker build -t ${env.DOCKER_IMAGE} ."
                }
            }
        }

        stage('Push to Docker Hub') {
            when {
                branch 'main'
            }
            steps {
                script {
                    withCredentials([usernamePassword(
                        credentialsId: 'docker-hub-credentials',
                        usernameVariable: 'DOCKER_USERNAME',
                        passwordVariable: 'DOCKER_PASSWORD'
                    )]) {
                        sh """
                            echo ${env.DOCKER_PASSWORD} | docker login -u ${env.DOCKER_USERNAME} --password-stdin
                            docker tag ${env.DOCKER_IMAGE} ${env.DOCKER_USERNAME}/${env.DOCKER_IMAGE}
                            docker tag ${env.DOCKER_IMAGE} ${env.DOCKER_USERNAME}/smarthealth-tracker:latest
                            docker push ${env.DOCKER_USERNAME}/${env.DOCKER_IMAGE}
                            docker push ${env.DOCKER_USERNAME}/smarthealth-tracker:latest
                        """
                    }
                }
            }
        }

        stage('Deploy with Docker Compose') {
            when {
                branch 'main'
            }
            steps {
                script {
                    sh 'docker-compose down || true'
                    sh 'docker-compose pull'
                    sh 'docker-compose up -d'
                    sh 'sleep 30'
                    sh 'docker-compose exec -T app php artisan migrate --force'
                }
            }
        }
    }

    post {
        always {
            echo '🧹 Cleaning up...'
            sh 'docker container prune -f'
            sh 'docker image prune -f'
            dir(env.LARAVEL_DIR) {
                sh 'rm -f database/database.sqlite || true'
            }
        }
        success {
            echo '✅ Pipeline completed successfully!'
        }
        failure {
            echo '❌ Pipeline failed!'
        }
    }
}
